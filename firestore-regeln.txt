rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Chaträume und Direktchats (dm_...)
    match /rooms/{room} {
      allow read: if request.auth != null;
      allow create: if request.auth != null
        && request.resource.data.name is string
        && request.resource.data.name.size() <= 40;
      match /msgs/{msg} {
        allow read: if request.auth != null;
        allow create: if request.auth != null
          && request.resource.data.uid == request.auth.uid
          && request.resource.size() < 20;
        allow update: if request.auth != null
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['poll', 'meet']);
        allow delete: if request.auth != null && resource.data.uid == request.auth.uid;
      }
    }
    // Öffentliches Profil (Nickname, Profilbild, Wohnort), ein Dokument pro Gerät
    match /users/{uid} {
      allow read: if request.auth != null;
      allow create, update: if request.auth != null && request.auth.uid == uid
        && request.resource.data.nick is string && request.resource.data.nick.size() <= 30
        && request.resource.data.av.size() < 60000;
      allow delete: if request.auth != null && request.auth.uid == uid;
    }
    // Übersicht der Direktchats
    match /dms/{dm} {
      allow read, delete: if request.auth != null;
      allow create, update: if request.auth != null
        && request.resource.data.members is list && request.resource.data.members.size() == 2;
    }
    // Genutzte E-Mail-Adressen (nur als Hash): jede Adresse nur einmal
    match /emails/{h} {
      allow get: if request.auth != null;
      allow create: if request.auth != null && h.size() == 64;
      allow list, update, delete: if false;
    }
    // Registrierungs-Anträge: anlegen ja, lesen nein
    match /pending/{id} {
      allow create: if request.auth != null && request.resource.data.id == id
        && request.resource.data.token is string && request.resource.data.token.size() >= 20
        && request.resource.data.hash is string && request.resource.data.hash.size() == 64
        && request.resource.data.name is string && request.resource.data.name.size() <= 30
        && !exists(/databases/$(database)/documents/accounts/$(id));
      allow read, update, delete: if false;
    }
    // Freigeschaltete Accounts: nur mit dem Schlüssel aus dem Freischalt-Link
    match /accounts/{id} {
      allow read: if request.auth != null;
      allow create: if request.auth != null
        && exists(/databases/$(database)/documents/pending/$(id))
        && request.resource.data.token == get(/databases/$(database)/documents/pending/$(id)).data.token
        && request.resource.data.hash == get(/databases/$(database)/documents/pending/$(id)).data.hash
        && request.resource.data.name == get(/databases/$(database)/documents/pending/$(id)).data.name
        && request.resource.data.g == get(/databases/$(database)/documents/pending/$(id)).data.g;
      allow update, delete: if false;
    }
    // Alles andere ist gesperrt
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
